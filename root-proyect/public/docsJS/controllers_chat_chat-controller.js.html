/**
 * Script de chat
 * Maneja:
 * - Polling periódico de mensajes nuevos
 * - Envío de mensajes al servidor
 * - Renderizado de burbujas de chat
 * - Scroll inteligente al área de mensajes
 * - Modo administrador con lista de conversaciones
 */

// Constantes
/** @constant {number} Intervalo de polling en milisegundos */
const POLLING_INTERVAL_MS = 3000;

/** @constant {number} Distancia en píxeles desde el fondo para activar el scroll automático */
const SCROLL_THRESHOLD_PX = 100;

// Estado del módulo 
/** @type {number} ID del último mensaje recibido */
let lastMessageId = 0;

/** @type {number|null} Referencia al temporizador de polling activo */
let pollingTimer = null;

/** @type {string|null} Tipo de sala activa ('admin' u otro) */
let activeRoomType = window.CHAT_ROOM_TYPE ?? null;

/** @type {number|null} ID de la sala activa */
let activeRoomId = window.CHAT_ROOM_ID ?? null;

/** @type {boolean} Indica si es la primera carga de mensajes (fuerza scroll al fondo) */
let isFirstLoad = true;

// Getters dinámicos
/** @returns {HTMLElement|null} Área de mensajes del chat */
const getMessagesArea = () => document.getElementById('chatMessagesArea');

/** @returns {HTMLFormElement|null} Formulario de envío del chat */
const getChatForm = () => document.getElementById('chatForm');

/** @returns {HTMLInputElement|null} Campo de texto del chat */
const getChatInput = () => document.getElementById('chatInput');

/** @returns {HTMLElement|null} Elemento de estado vacío del chat */
const getEmptyState = () => document.getElementById('chatEmptyState');

/**
 * Inicializa el controlador del chat.
 * Lee la configuración global de la ventana, arranca el polling o el modo admin
 * según corresponda, y registra los listeners de eventos.
 */
function init() {
    const form = getChatForm();
    const input = getChatInput();

    if (!form || !input) return;

    activeRoomType = window.CHAT_ROOM_TYPE ?? activeRoomType;
    activeRoomId = window.CHAT_ROOM_ID ?? activeRoomId;

    if (window.CHAT_ADMIN_MODE) {
        initAdminMode();
    } else {
        startPolling(activeRoomType, activeRoomId);
    }

    form.addEventListener('submit', handleSendMessage);
    window.addEventListener('beforeunload', stopPolling);
}

if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
} else {
    init();
}

/** Expone init globalmente para reinicializar el chat desde otros módulos */
window.initChatController = init;

/**
 * Inicia el polling de mensajes para una sala concreta.
 * Detiene cualquier polling previo antes de arrancar el nuevo.
 *
 * @param {string} roomType - Tipo de sala ('admin' u otro valor definido por el servidor)
 * @param {number} roomId   - ID de la sala o usuario destinatario
 */
function startPolling(roomType, roomId) {
    stopPolling();

    activeRoomType = roomType;
    activeRoomId = roomId;
    lastMessageId = 0;
    isFirstLoad = true;

    fetchNewMessages();
    pollingTimer = setInterval(fetchNewMessages, POLLING_INTERVAL_MS);
}

/**
 * Detiene el polling activo si existe.
 */
function stopPolling() {
    if (pollingTimer !== null) {
        clearInterval(pollingTimer);
        pollingTimer = null;
    }
}

/**
 * Solicita al servidor los mensajes posteriores al último ID recibido.
 * Si hay mensajes nuevos los renderiza y actualiza el estado del módulo.
 *
 * @returns {Promise<void>}
 */
async function fetchNewMessages() {
    if (!activeRoomId) return;

    const url = new URL('index.php', window.location.origin + window.location.pathname.replace('index.php', ''));
    url.searchParams.set('accion', 'getMessages');
    url.searchParams.set('room_type', activeRoomType);
    url.searchParams.set('room_id', activeRoomId);
    url.searchParams.set('after_id', lastMessageId);

    try {
        const response = await fetch(url.toString());
        const data = await response.json();

        if (!data.success) return;

        if (data.messages.length > 0) {
            renderMessages(data.messages);
            lastMessageId = data.last_id;
            hideEmptyState();
        }

    } catch (error) {
        showAlert('Error de red', 'Se ha producido un error', 'error');
    }
}

/**
 * Maneja el envío del formulario de chat.
 * Valida el texto, lo envía al servidor y renderiza la respuesta.
 *
 * @param {SubmitEvent} event - Evento de envío del formulario
 * @returns {Promise<void>}
 */
async function handleSendMessage(event) {
    event.preventDefault();

    const input = getChatInput();
    if (!input) return;

    const text = input.value.trim();
    if (!text || !activeRoomId) return;

    input.disabled = true;

    try {
        const response = await fetch('index.php?accion=sendMessage', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                room_type: activeRoomType,
                room_id: activeRoomId,
                message: text,
            }),
        });

        const data = await response.json();

        if (data.success) {
            renderMessages([data.message]);
            scrollToBottomIfNeeded(true);
            lastMessageId = data.message.id;
            hideEmptyState();
            input.value = '';
        } else {
            showAlert('Error', 'No se pudo enviar el mensaje. Inténtalo de nuevo.', 'error');
        }

    } catch (error) {
        showAlert('Error de red', 'Se ha producido un error', 'error');
    } finally {
        input.disabled = false;
        input.focus();
    }
}

/**
 * Renderiza una lista de mensajes en el área del chat.
 * Omite mensajes ya presentes en el DOM e ignora duplicados.
 * En la primera carga fuerza el scroll al fondo; en las siguientes
 * aplica el scroll inteligente.
 *
 * @param {Object[]} messages            - Lista de mensajes a renderizar
 * @param {number}   messages[].id        - ID único del mensaje
 * @param {number}   messages[].sender_id - ID del usuario que envió el mensaje
 * @param {string}   messages[].message   - Texto del mensaje
 * @param {string}   messages[].created_at - Fecha/hora de creación (formato MySQL)
 * @param {string}   messages[].sender_name - Nombre del remitente
 */
function renderMessages(messages) {
    const area = getMessagesArea();
    if (!area) return;

    const currentUserId = window.CURRENT_USER?.id;
    const fragment = document.createDocumentFragment();

    messages.forEach(msg => {
        if (document.getElementById(`chat-msg-${msg.id}`)) return;

        const isOwn = parseInt(msg.sender_id) === parseInt(currentUserId);
        fragment.appendChild(buildMessageBubble(msg, isOwn));
    });

    area.appendChild(fragment);

    if (isFirstLoad) {
        scrollToBottomIfNeeded(true);
        isFirstLoad = false;
        return;
    }

    scrollToBottomIfNeeded();
}

/**
 * Construye el elemento DOM de una burbuja de mensaje.
 *
 * @param {Object}  msg             - Datos del mensaje
 * @param {number}  msg.id          - ID único del mensaje
 * @param {string}  msg.sender_name - Nombre del remitente
 * @param {string}  msg.message     - Texto del mensaje
 * @param {string}  msg.created_at  - Fecha/hora de creación (formato MySQL)
 * @param {boolean} isOwn           - true si el mensaje pertenece al usuario actual
 * @returns {HTMLDivElement} Elemento wrapper de la burbuja listo para insertar en el DOM
 */
function buildMessageBubble(msg, isOwn) {
    const wrapper = document.createElement('div');
    wrapper.id = `chat-msg-${msg.id}`;
    wrapper.className = `chat-bubble ${isOwn ? 'chat-bubble--own' : 'chat-bubble--other'}`;

    const senderEl = document.createElement('span');
    senderEl.className = 'chat-bubble__sender';
    senderEl.textContent = isOwn ? 'Tú' : msg.sender_name;

    const textEl = document.createElement('p');
    textEl.className = 'chat-bubble__text';
    textEl.textContent = msg.message;

    const timeEl = document.createElement('time');
    timeEl.className = 'chat-bubble__time';
    timeEl.textContent = formatTime(msg.created_at);

    const bodyEl = document.createElement('div');
    bodyEl.className = 'chat-bubble__body';
    bodyEl.appendChild(textEl);
    bodyEl.appendChild(timeEl);

    wrapper.appendChild(senderEl);
    wrapper.appendChild(bodyEl);

    return wrapper;
}

/**
 * Formatea una fecha en cadena a hora local HH:MM.
 *
 * @param {string} dateStr - Fecha en formato MySQL ('YYYY-MM-DD HH:MM:SS')
 * @returns {string} Hora formateada, por ejemplo '14:05'
 */
function formatTime(dateStr) {
    const date = new Date(dateStr.replace(' ', 'T'));
    return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
}

/**
 * Desplaza el área de mensajes al fondo, con comportamiento inteligente opcional.
 * Si force es true, siempre hace scroll.
 * Si force es false, solo hace scroll cuando el usuario está cerca del fondo.
 *
 * @param {boolean} [force=false] - Fuerza el scroll al fondo independientemente de la posición actual
 */
function scrollToBottomIfNeeded(force = false) {
    const area = getMessagesArea();
    if (!area) return;

    if (force) {
        area.scrollTop = area.scrollHeight;
        return;
    }

    const distanceFromBottom = area.scrollHeight - area.scrollTop - area.clientHeight;

    if (distanceFromBottom <= SCROLL_THRESHOLD_PX) {
        area.scrollTop = area.scrollHeight;
    }
}

/**
 * Oculta el estado vacío del chat si está visible.
 */
function hideEmptyState() {
    const el = getEmptyState();
    if (el) el.style.display = 'none';
}

/**
 * Inicializa el modo administrador.
 * Carga la lista de conversaciones, preselecciona un usuario si viene indicado
 * en la configuración global y activa la recarga periódica de conversaciones.
 *
 * @returns {Promise<void>}
 */
async function initAdminMode() {
    await loadAdminConversations();

    if (window.CHAT_PRESELECT_USER_ID) {
        const targetItem = document.querySelector(`[data-user-id="${window.CHAT_PRESELECT_USER_ID}"]`);
        if (targetItem) {
            const fullName = targetItem.querySelector('.chat-user-item__name')?.textContent || 'Usuario';
            openAdminConversation(window.CHAT_PRESELECT_USER_ID, fullName);
        }
    }

    setInterval(loadAdminConversations, 10000);
}

/**
 * Solicita al servidor la lista de salas de chat y las renderiza.
 *
 * @returns {Promise<void>}
 */
async function loadAdminConversations() {
    try {
        const response = await fetch('index.php?accion=getChatRooms');
        const data = await response.json();

        if (!data.success) {
            showAlert('Error', 'No se pudo cargar las conversaciones correctamente.', 'error');
            return;
        }

        renderConversationList(data.conversations);

    } catch (error) {
        showAlert('Error de red', 'Se ha producido un error', 'error');
    }
}

/**
 * Renderiza la lista de conversaciones en el panel de administración.
 * Añade ítems nuevos y actualiza la previsualización de los existentes.
 *
 * @param {Object[]}      conversations                - Lista de conversaciones
 * @param {number}        conversations[].user_id      - ID del usuario de la conversación
 * @param {string}        conversations[].full_name    - Nombre completo del usuario
 * @param {string|null}   conversations[].last_message - Último mensaje de la conversación
 * @param {string|null}   conversations[].last_message_at - Fecha del último mensaje (formato MySQL)
 */
function renderConversationList(conversations) {
    const list = document.getElementById('chatUserList');
    const emptyItem = document.getElementById('chatUserListEmpty');

    if (!list) return;

    if (conversations.length === 0) {
        if (emptyItem) emptyItem.style.display = 'flex';
        return;
    }

    if (emptyItem) emptyItem.style.display = 'none';

    const existingIds = new Set(
        [...list.querySelectorAll('[data-user-id]')].map(el => el.dataset.userId)
    );

    conversations.forEach(conv => {
        const strId = String(conv.user_id);

        if (!existingIds.has(strId)) {
            list.appendChild(buildConversationItem(conv));
        } else {
            const existingItem = list.querySelector(`[data-user-id="${strId}"]`);
            if (existingItem && conv.last_message) {
                const preview = existingItem.querySelector('.chat-user-item__preview');
                if (preview) preview.textContent = conv.last_message;
            }
        }
    });
}

/**
 * Construye el elemento DOM de un ítem de conversación para el panel admin.
 *
 * @param {Object}      conv                - Datos de la conversación
 * @param {number}      conv.user_id        - ID del usuario
 * @param {string}      conv.full_name      - Nombre completo del usuario
 * @param {string|null} conv.last_message   - Texto del último mensaje
 * @param {string|null} conv.last_message_at - Fecha del último mensaje (formato MySQL)
 * @returns {HTMLLIElement} Elemento de lista listo para insertar en el DOM
 */
function buildConversationItem(conv) {
    const li = document.createElement('li');
    li.className = 'chat-user-item';
    li.dataset.userId = conv.user_id;
    li.setAttribute('role', 'button');
    li.setAttribute('tabindex', '0');
    li.setAttribute('aria-label', `Chat con ${conv.full_name}`);

    li.innerHTML = `
        <div class="chat-user-item__info">
            <strong class="chat-user-item__name">${escapeHtml(conv.full_name)}</strong>
            <span class="chat-user-item__preview">${escapeHtml(conv.last_message ?? '')}</span>
        </div>
        <time class="chat-user-item__time">${conv.last_message_at ? formatTime(conv.last_message_at) : ''}</time>
    `;

    const selectConversation = () => openAdminConversation(conv.user_id, conv.full_name);
    li.addEventListener('click', selectConversation);
    li.addEventListener('keydown', (e) => { if (e.key === 'Enter') selectConversation(); });

    return li;
}

/**
 * Abre la conversación de un usuario en el panel admin.
 * Actualiza el ítem activo en la lista, limpia el área de mensajes
 * y arranca el polling para esa conversación.
 *
 * @param {number} userId   - ID del usuario cuya conversación se abre
 * @param {string} fullName - Nombre completo del usuario, mostrado en el encabezado
 */
function openAdminConversation(userId, fullName) {
    document.querySelectorAll('.chat-user-item').forEach(el => el.classList.remove('chat-user-item--active'));

    const activeItem = document.querySelector(`[data-user-id="${userId}"]`);
    if (activeItem) activeItem.classList.add('chat-user-item--active');

    const nameEl = document.getElementById('chatSelectedUserName');
    if (nameEl) nameEl.textContent = fullName;

    const area = getMessagesArea();
    if (area) area.innerHTML = '';

    isFirstLoad = true;

    showConversationPanel();
    startPolling('admin', userId);
}

/**
 * Muestra el panel de conversación ocultando la pantalla de selección vacía.
 */
function showConversationPanel() {
    const noSelection = document.getElementById('chatNoSelection');
    const header = document.getElementById('chatConversationHeader');
    const form = getChatForm();
    const area = getMessagesArea();

    if (noSelection) noSelection.style.display = 'none';
    if (header) header.classList.remove('chat-header--hidden');
    if (area) area.classList.remove('chat-messages-area--hidden');
    if (form) form.classList.remove('chat-input-area--hidden');
}
